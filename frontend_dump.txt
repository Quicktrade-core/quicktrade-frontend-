// src/App.jsx
import { createBrowserRouter, RouterProvider } from "react-router"
import ContextProvider from "./context/contextProvider"
import RootLayout from "./rootLayout"
import HomePage from "./pages/homePage"
import CartPage from "./pages/cartPage"
import NotfoundPage from "./pages/notfoundPage"
import AuthPage from "./pages/authPage"
import RegisterPage from "./pages/registerPage"
import VideosPage from "./pages/videosPage"
import ProductPage from "./pages/productPage"
import CheckoutPage from "./pages/checkoutPage"
import AccountPage from "./pages/accountPage/accountPage"
import ProfilePage from "./pages/accountPage/profilePage"
import WalletPage from "./pages/accountPage/walletPage"
import NotificationsPage from "./pages/accountPage/notificationsPage"
import SecurityPage from "./pages/accountPage/securityPage"
import SettingsPage from "./pages/accountPage/settingsPage"
import EbooksPage from "./pages/accountPage/ebooksPage"
import TicketsPage from "./pages/accountPage/ticketsPage"
import TicketDetailPage from "./pages/ticketDetailPage"
import OrdersPage from "./pages/accountPage/ordersPage"
import OrderPage from "./pages/accountPage/orderPage"
import SavedPage from "./pages/accountPage/savedPage"
import SellerPage from "./pages/sellerPage"
import Products from "./components/seller/products"
import Reels from "./components/seller/reels"
import Stores from "./components/seller/stores"
import ForgotPasswordPage from "./pages/forgotPasswordPage/forgotPasswordPage"
import VerificationPage from "./pages/forgotPasswordPage/verificationPage"
import NewPasswordPage from "./pages/forgotPasswordPage/newPasswordPage"
import SearchPage from "./pages/searchPage"

function App() {
  const router = createBrowserRouter([
    { path: "/", element: <RootLayout />, children: [{ index: true, element: <HomePage /> }] },
    {
      path: "/seller",
      element: <SellerPage />,
      children: [
        { index: true, element: <Products /> },
        { path: "reels", element: <Reels /> },
        { path: "stores", element: <Stores /> },
      ],
    },
    { path: "/search", element: <SearchPage /> },
    { path: "/product/:id", element: <ProductPage /> },
    { path: "/cart", element: <CartPage /> },
    { path: "/checkout", element: <CheckoutPage /> },
    { path: "/videos", element: <VideosPage /> },
    { path: "/account", element: <AccountPage /> },
    { path: "/profile", element: <ProfilePage /> },
    { path: "/wallet", element: <WalletPage /> },
    { path: "/notifications", element: <NotificationsPage /> },
    { path: "/security", element: <SecurityPage /> },
    { path: "/settings", element: <SettingsPage /> },
    { path: "/ebooks", element: <EbooksPage /> },
    { path: "/tickets", element: <TicketsPage /> },
    { path: "/tickets/:id", element: <TicketDetailPage /> },
    { path: "/orders", element: <OrdersPage /> },
    { path: "/order/:id", element: <OrderPage /> },
    { path: "/saved", element: <SavedPage /> },
    { path: "/auth", element: <AuthPage /> },
    { path: "/register", element: <RegisterPage /> },
    { path: "/forgotpassword", element: <ForgotPasswordPage /> },
    { path: "/verification", element: <VerificationPage /> },
    { path: "/newpassword", element: <NewPasswordPage /> },
    { path: "*", element: <NotfoundPage /> },
  ])

  return (
    <ContextProvider>
      <RouterProvider router={router} />
    </ContextProvider>
  )
}

export default App

// src/components/cartItem.jsx
import { useContext, useState } from "react"
import { cartContext } from "../context/cartContext"
import { BiMinus, BiPlus, BiX } from "react-icons/bi"
import { Link } from "react-router"
import FormatPrice from "../utils/formatPrice"
import { FaNairaSign } from "react-icons/fa6"

const CartItem = ({ item }) => {
  const { updateQuantity, removeFromCart } = useContext(cartContext)
  const [busy, setBusy] = useState(false)

  const listing = item.listing
  const lineTotal = parseFloat(item.line_buyer_total || 0)

  const changeQty = async (newQty) => {
    if (busy) return
    setBusy(true)
    await updateQuantity(item.cart_item_id, newQty)
    setBusy(false)
  }

  const handleRemove = async () => {
    if (busy) return
    setBusy(true)
    await removeFromCart(item.cart_item_id)
    setBusy(false)
  }

  return (
    <div className={`flex gap-5 ${busy ? "opacity-60" : ""}`}>
      <Link to={`/product/${listing.id}`} className="w-1/3 h-20">
        <img src={listing.image_url || "/placeholder.png"} alt={listing.title} className="w-full h-full object-cover rounded-xl" />
      </Link>
      <div className="w-2/3 flex flex-col justify-between">
        <div className="flex justify-between items-center gap-5">
          <p className="leading-tight truncate">{listing.title}</p>
          <button className="cursor-pointer" onClick={handleRemove} disabled={busy}>
            <BiX className="text-2xl" />
          </button>
        </div>
        <div className="flex justify-between items-center">
          <div className="w-1/2 flex flex-col">
            <p className="font-semibold flex items-center">
              <FaNairaSign />{FormatPrice(parseFloat(listing.buyer_price || listing.price || 0))}
            </p>
            {item.quantity > 1 && (
              <p className="text-xs text-neutral-500 flex items-center gap-1">
                Total: <FaNairaSign className="text-[10px]" />{FormatPrice(lineTotal)}
              </p>
            )}
          </div>
          <div className="flex justify-between items-center border-2 border-neutral-200 dark:border-neutral-700 rounded-full w-1/2 px-2">
            <button className="cursor-pointer" onClick={() => changeQty(item.quantity - 1)} disabled={busy}>
              <BiMinus />
            </button>
            {item.quantity}
            <button className="cursor-pointer" onClick={() => changeQty(item.quantity + 1)} disabled={busy}>
              <BiPlus />
            </button>
          </div>
        </div>
      </div>
    </div>
  )
}

export default CartItem

import { useEffect, useRef, useState } from "react"
import { Link } from "react-router"
import { FaCalendarAlt, FaMapMarkerAlt, FaVolumeMute, FaVolumeUp } from "react-icons/fa"
import { FaNairaSign } from "react-icons/fa6"
import FormatPrice from "../utils/formatPrice"

const HeroBanner = ({ slides }) => {
  const [current, setCurrent] = useState(0)
  const [showCard, setShowCard] = useState(false)
  const [isMuted, setIsMuted] = useState(true)
  const foregroundRef = useRef(null)
  const backdropRef = useRef(null)
  const cardTimeoutRef = useRef(null)

  const active = slides[current]

  useEffect(() => {
    return () => {
      if (cardTimeoutRef.current) clearTimeout(cardTimeoutRef.current)
    }
  }, [])

  const handleVideoEnd = () => {
    setShowCard(true)
    cardTimeoutRef.current = setTimeout(() => {
      setShowCard(false)
      setCurrent((prev) => (prev + 1) % slides.length)
    }, 5000)
  }

  useEffect(() => {
    const fg = foregroundRef.current
    const bg = backdropRef.current

    if (fg) {
      try {
        fg.currentTime = 0
        fg.muted = isMuted
        fg.play().catch(() => {})
      } catch (err) {
        console.warn("Foreground video restart failed:", err)
      }
    }

    if (bg) {
      try {
        bg.currentTime = 0
        bg.play().catch(() => {})
      } catch (err) {
        console.warn("Backdrop video restart failed:", err)
      }
    }
  }, [current, isMuted])

  const toggleMute = () => {
    setIsMuted((prev) => !prev)
  }

  if (!active) return null

  const ticket = active.ticket

  return (
    <div className="h-[35vh] mx-4 rounded-2xl relative overflow-hidden bg-neutral-900">
      {/* LAYER 1: blurred backdrop */}
      <video
        ref={backdropRef}
        className="absolute inset-0 w-full h-full object-cover scale-110 blur-lg"
        src={active.video}
        autoPlay
        muted
        loop
        playsInline
        preload="metadata"
        poster={active.poster}
      />

      {/* LAYER 2: dark gradient tint */}
      <div className="absolute inset-0 bg-gradient-to-b from-black/20 via-black/10 to-black/40"></div>

      {/* LAYER 3: foreground 9:16 video, centered */}
      <div
        className={`absolute inset-0 flex items-center justify-center transition-all duration-500 ${
          showCard ? "blur-sm" : ""
        }`}
      >
        <video
          ref={foregroundRef}
          className="h-full w-auto aspect-[9/16] object-cover rounded-2xl shadow-2xl"
          src={active.video}
          autoPlay
          muted={isMuted}
          playsInline
          preload="metadata"
          poster={active.poster}
          onEnded={handleVideoEnd}
        />
      </div>

      {/* Mute toggle */}
      <button
        onClick={toggleMute}
        className="absolute top-3 right-3 z-20 bg-black/50 hover:bg-black/70 text-white p-2 rounded-full cursor-pointer transition-colors"
        aria-label={isMuted ? "Unmute video" : "Mute video"}
      >
        {isMuted ? <FaVolumeMute className="text-sm" /> : <FaVolumeUp className="text-sm" />}
      </button>

      {/* LAYER 4: ticket overlay — horizontal glass card */}
      {showCard && ticket && (
        <div className="absolute inset-0 flex items-center justify-center px-4 z-10 pointer-events-none">
          <div className="bg-black/70 backdrop-blur-md rounded-2xl px-5 py-4 flex items-center gap-4 max-w-md pointer-events-auto">
            <img
              src={ticket.image}
              alt={ticket.name}
              className="w-20 h-20 rounded-xl object-cover shadow-lg shrink-0"
            />

            <div className="flex flex-col gap-1.5 min-w-0 flex-1">
              <p className="text-white font-bold text-sm leading-tight truncate">
                {ticket.name}
              </p>

              {ticket.tier && (
                <span className="bg-white/20 text-white text-[10px] font-semibold px-2 py-0.5 rounded-full self-start">
                  {ticket.tier}
                </span>
              )}

              <div className="flex items-center gap-3 text-white/80 text-[11px] flex-wrap">
                {ticket.date && (
                  <span className="flex items-center gap-1">
                    <FaCalendarAlt /> {ticket.date}
                  </span>
                )}
                {ticket.venue && (
                  <span className="flex items-center gap-1 truncate">
                    <FaMapMarkerAlt /> {ticket.venue}
                  </span>
                )}
              </div>

              <Link
                to={`/product/${ticket.id}`}
                className="mt-1 bg-white text-neutral-900 hover:bg-neutral-100 px-4 py-1.5 rounded-full font-semibold text-xs flex items-center justify-center gap-1 self-start transition-colors"
              >
                Buy now · <FaNairaSign className="text-[10px]" />
                {FormatPrice(parseFloat(ticket.price || 0))}
              </Link>
            </div>
          </div>
        </div>
      )}

      {/* LAYER 5: dots */}
      <div className="absolute bottom-2 left-1/2 -translate-x-1/2 flex gap-1.5 z-10">
        {slides.map((_, i) => (
          <span
            key={i}
            className={`h-1.5 rounded-full transition-all ${

            }`}

import { createContext, useCallback, useEffect, useState } from "react";
import { api, TOKEN_STORAGE_KEY } from "../lib/api";

export const authContext = createContext();

const AuthProvider = ({ children }) => {
  const [user, setUser] = useState(null);
  const [isAuthenticated, setIsAuthenticated] = useState(false);
  const [loading, setLoading] = useState(true);

  // Kept so existing toggle UI still works (login vs signup view)
  const [login, setLogin] = useState(true);

  const persistUser = useCallback((nextUser) => {
    setUser(nextUser);
    setIsAuthenticated(!!nextUser);
  }, []);

  const clearSession = useCallback(() => {
    localStorage.removeItem(TOKEN_STORAGE_KEY);
    setUser(null);
    setIsAuthenticated(false);
  }, []);

  const checkAuth = useCallback(async () => {
    const token = localStorage.getItem(TOKEN_STORAGE_KEY);
    if (!token) {
      setLoading(false);
      return;
    }
    try {
      const res = await api.get("/profile/");
      if (res.data?.success) persistUser(res.data.data);
      else clearSession();
    } catch {
      clearSession();
    } finally {
      setLoading(false);
    }
  }, [persistUser, clearSession]);

  // ---- Register ----
  const registerUser = async ({ email, password, password2, first_name, last_name, phone, current_mode }) => {
    try {
      const res = await api.post("/accounts/register/", {
        email,
        password,
        password2: password2 || password,
        first_name: first_name || "",
        last_name: last_name || "",
        phone: phone || "",
        current_mode: current_mode || "Buyer",
      });
      if (res.data?.success) {
        return { success: true, data: res.data.data };
      }
      return { success: false, error: res.data?.error || "Registration failed" };
    } catch (err) {
      return { success: false, error: err.response?.data?.error || "Registration failed" };
    }
  };

  // ---- Resend OTP ----
  const resendOTP = async (email) => {
    try {
      const res = await api.post("/accounts/resend-otp/", { email });
      if (res.data?.success) return { success: true, message: res.data.message };
      return { success: false, error: res.data?.error || "Failed to resend OTP" };
    } catch (err) {
      return { success: false, error: err.response?.data?.error || "Failed to resend OTP" };
    }
  };

  // ---- Verify OTP ----
  const verifyOTP = async ({ email, otp }) => {
    try {
      const res = await api.post("/accounts/verify-otp/", { email, otp });
      if (res.data?.success) {
        const { user: verified, token } = res.data.data || {};
        if (token) localStorage.setItem(TOKEN_STORAGE_KEY, token);
        if (verified) persistUser(verified);
        else await checkAuth();
        return { success: true, data: res.data.data };
      }
      return { success: false, error: res.data?.error || "OTP verification failed" };
    } catch (err) {
      return {
        success: false,
        error: err.response?.data?.error || "OTP verification failed",
        code: err.response?.data?.code,
      };
    }
  };

  // ---- Login ----
  const loginUser = async ({ identifier, password, remember_me = false }) => {
    try {
      const res = await api.post("/accounts/sign-in/", { identifier, password, remember_me });
      if (res.data?.success) {
        const { token, user: signedIn } = res.data.data || {};
        if (token) localStorage.setItem(TOKEN_STORAGE_KEY, token);
        if (signedIn) persistUser(signedIn);
        else await checkAuth();
        return { success: true, data: res.data.data };
      }
      return { success: false, error: res.data?.error || "Login failed" };
    } catch (err) {
      return {
        success: false,
        error: err.response?.data?.error || "Login failed",
        code: err.response?.data?.code,
      };
    }
  };

  // ---- Logout ----
  const logoutUser = async () => {
    try {
      await api.post("/accounts/logout/");
    } catch (err) {
      console.error("Logout error:", err);
    } finally {
      clearSession();
    }
  };

  // ---- Select mode ----
  const updateUserMode = async (mode, gender = "Male") => {
    try {
      const res = await api.post("/accounts/select-mode/", { mode, gender });
      if (res.data?.success) {
        setUser((prev) => (prev ? { ...prev, current_mode: res.data.data.current_mode, gender: res.data.data.gender } : prev));
        return { success: true, data: res.data.data };
      }
      return { success: false, error: res.data?.error || "Failed to update mode" };
    } catch (err) {
      return { success: false, error: err.response?.data?.error || "Failed to update mode" };
    }
  };

  // ---- Profile ----
  const refreshUser = useCallback(async () => {
    try {
      const res = await api.get("/profile/");
      if (res.data?.success) {
        persistUser(res.data.data);
        return { success: true, data: res.data.data };
      }
      return { success: false, error: res.data?.error || "Failed to refresh" };
    } catch (err) {
      return { success: false, error: err.response?.data?.error || "Failed to refresh" };
    }
  }, [persistUser]);

  const updateProfile = async (payload) => {
    try {
      const res = await api.post("/profile/edit/", payload);
      if (res.data?.success) {
        persistUser(res.data.data);
        return { success: true, data: res.data.data };
      }
      return { success: false, error: res.data?.error || "Update failed" };
    } catch (err) {
      return { success: false, error: err.response?.data?.error || "Update failed" };
    }
  };

  const uploadAvatar = async (file) => {
    try {
      const fd = new FormData();
      fd.append("avatar", file);
      const res = await api.post("/profile/avatar/", fd, {
        headers: { "Content-Type": "multipart/form-data" },
      });
      if (res.data?.success) {
        setUser((prev) =>
          prev ? { ...prev, profile: { ...prev.profile, avatar_url: res.data.data.avatar_url } } : prev
        );
        return { success: true, data: res.data.data };
      }
      return { success: false, error: res.data?.error || "Upload failed" };
    } catch (err) {
      return { success: false, error: err.response?.data?.error || "Upload failed" };
    }
  };

  useEffect(() => {
    checkAuth();
  }, [checkAuth]);

  return (
    <authContext.Provider
      value={{
        user,
        isAuthenticated,
        loading,
        login,
        setLogin,
        api,
        registerUser,
        resendOTP,
        verifyOTP,
        loginUser,
        logoutUser,
        updateUserMode,
        refreshUser,
        updateProfile,
        uploadAvatar,
        checkAuth,
      }}
    >
      {children}
    </authContext.Provider>
  );
};

import { createContext, useCallback, useContext, useEffect, useState } from "react"
import { authContext } from "./authContext"

export const cartContext = createContext()

const CartProvider = ({ children }) => {
  const { api, isAuthenticated } = useContext(authContext)

  const [cartItems, setCartItems] = useState([])
  const [subtotal, setSubtotal] = useState("0.00")
  const [buyerFeesTotal, setBuyerFeesTotal] = useState("0.00")
  const [delivery, setDelivery] = useState("0.00")
  const [grandTotal, setGrandTotal] = useState("0.00")
  const [cartCount, setCartCount] = useState(0)
  const [ticketCount, setTicketCount] = useState(0)
  const [loading, setLoading] = useState(false)
  const [error, setError] = useState("")

  const refreshCart = useCallback(async () => {
    if (!isAuthenticated) {
      setCartItems([])
      setSubtotal("0.00"); setBuyerFeesTotal("0.00"); setDelivery("0.00"); setGrandTotal("0.00")
      setCartCount(0)
      return
    }
    setLoading(true); setError("")
    try {
      const res = await api.get("/transactions/cart/")
      if (res.data?.success) {
        setCartItems(res.data.items || [])
        setSubtotal(res.data.subtotal || "0.00")
        setBuyerFeesTotal(res.data.buyer_fees_total || "0.00")
        setDelivery(res.data.delivery || "0.00")
        setGrandTotal(res.data.grand_total || "0.00")
        setCartCount((res.data.items || []).reduce((s, i) => s + (i.quantity || 0), 0))
      } else {
        setError(res.data?.error || "Failed to load cart")
      }
    } catch (err) {
      if (err.response?.status !== 401) setError("Network error loading cart")
      else {
        setCartItems([]); setCartCount(0)
        setSubtotal("0.00"); setBuyerFeesTotal("0.00"); setDelivery("0.00"); setGrandTotal("0.00")
      }
    } finally {
      setLoading(false)
    }
  }, [api, isAuthenticated])

  const refreshTicketCount = useCallback(async () => {
    if (!isAuthenticated) { setTicketCount(0); return }
    try {
      const res = await api.get("/transactions/tickets/count/")
      if (res.data?.success) setTicketCount(res.data.data.active_tickets || 0)
    } catch {
      setTicketCount(0)
    }
  }, [api, isAuthenticated])

  useEffect(() => {
    refreshCart()
    refreshTicketCount()
  }, [refreshCart, refreshTicketCount])

  const addToCart = async (listingId, quantity = 1) => {
    if (!isAuthenticated) return { success: false, error: "Please sign in first.", code: "NOT_AUTHENTICATED" }
    try {
      const res = await api.post(`/transactions/cart/add/${listingId}/`, { quantity })
      if (res.data?.success) await refreshCart()
      return { success: res.data?.success, data: res.data }
    } catch (err) {
      return { success: false, error: err.response?.data?.error || "Could not add to cart" }
    }
  }

  const updateQuantity = async (cartItemId, quantity) => {
    try {
      const res = await api.post(`/transactions/cart/update/${cartItemId}/`, { quantity })
      if (res.data?.success) await refreshCart()
      return { success: res.data?.success, data: res.data }
    } catch (err) {
      return { success: false, error: err.response?.data?.error || "Could not update quantity" }
    }
  }

  const removeFromCart = async (cartItemId) => {
    try {
      const res = await api.post(`/transactions/cart/remove/${cartItemId}/`)
      if (res.data?.success) await refreshCart()
      return { success: res.data?.success, data: res.data }
    } catch (err) {
      return { success: false, error: err.response?.data?.error || "Could not remove item" }
    }
  }

  const clearCart = async () => {
    try {
      const res = await api.post("/transactions/cart/clear/")
      if (res.data?.success) await refreshCart()
      return { success: res.data?.success, data: res.data }
    } catch (err) {
      return { success: false, error: err.response?.data?.error || "Could not clear cart" }
    }
  }

  return (
    <cartContext.Provider value={{
      cartItems, subtotal, buyerFeesTotal, delivery, grandTotal,
      cartCount, ticketCount,
      loading, error,
      addToCart, updateQuantity, removeFromCart, clearCart,
      refreshCart, refreshTicketCount,
    }}>
      {children}
    </cartContext.Provider>
  )
}

      setError(err.response?.data?.error || "Network error")
    } finally {
      setLoading(false)
    }
  }, [api, isAuthenticated])

  useEffect(() => { refreshTickets() }, [refreshTickets])

  return (
    <ticketContext.Provider value={{ tickets, setTickets, loading, error, refreshTickets }}>
      {children}
    </ticketContext.Provider>

import { useEffect, useState } from "react"

export const useDebouncedValue = (value, delayMs = 300) => {
  const [debounced, setDebounced] = useState(value)

  useEffect(() => {
    const timer = setTimeout(() => setDebounced(value), delayMs)
    return () => clearTimeout(timer)
  }, [value, delayMs])

  return debounced
}

// src/lib/api.js
import axios from "axios";

export const API_BASE_URL =
  "https://quicktrade-backend.onrender.com";

const STORAGE_KEY = "authToken";

export const api = axios.create({
  baseURL: API_BASE_URL,
  headers: { "Content-Type": "application/json" },
  withCredentials: true,
});

// Send `Token <key>` on every request — matches the backend's api_login_required
api.interceptors.request.use(
  (config) => {
    const token = localStorage.getItem(STORAGE_KEY);
    if (token) config.headers.Authorization = `Token ${token}`;
    return config;
  },
  (error) => Promise.reject(error)
);

// On 401, clear the token and bounce to /auth
api.interceptors.response.use(
  (response) => response,
  async (error) => {
    if (error.response?.status === 401) {
      localStorage.removeItem(STORAGE_KEY);
      if (window.location.pathname !== "/auth") {
        window.location.href = "/auth";
      }
    }
    return Promise.reject(error);
  }
);

export const TOKEN_STORAGE_KEY = STORAGE_KEY;

      day: "numeric",
      year: "numeric",
      hour: "2-digit",
      minute: "2-digit",

  } catch {
    return dateStr

}

// src/pages/forgotPasswordPage/verificationPage.jsx
import { useContext, useEffect, useRef, useState } from "react"
import { useNavigate, useSearchParams } from "react-router"
import { FaArrowLeft } from "react-icons/fa"
import { authContext } from "../../context/authContext"
import Scrolltotop from "../../utils/scrolltotop"
import verify from "../../assets/verify.png"

const VerificationPage = () => {
  const navigate = useNavigate()
  const [params] = useSearchParams()

  const {
    verifyOTP,
    isAuthenticated,
    loading: authLoading,
    api,
  } = useContext(authContext)

  // Which flow are we in?
  //   ?email=...           -> registration OTP
  //   ?email=...&mode=reset -> password reset OTP
  const email = params.get("email") || ""
  const mode = params.get("mode") || "registration"

  const [digits, setDigits] = useState(["", "", "", "", "", ""])
  const [loading, setLoading] = useState(false)
  const [resending, setResending] = useState(false)
  const [error, setError] = useState("")
  const [info, setInfo] = useState("")

  const inputsRef = useRef([])

  // If already signed in (e.g. registration completed elsewhere), skip
  useEffect(() => {
    if (mode === "registration" && !authLoading && isAuthenticated) {
      navigate("/")
    }
  }, [authLoading, isAuthenticated, mode, navigate])

  // Focus first box on mount
  useEffect(() => {
    inputsRef.current[0]?.focus()
  }, [])

  const otp = digits.join("")

  // ---- Box handlers ----

  const handleChange = (index, value) => {
    const v = value.replace(/\D/g, "").slice(-1)
    const next = [...digits]
    next[index] = v
    setDigits(next)
    if (v && index < 5) inputsRef.current[index + 1]?.focus()
  }

  const handleKeyDown = (index, e) => {
    if (e.key === "Backspace" && !digits[index] && index > 0) {
      inputsRef.current[index - 1]?.focus()
    }
  }

  const handlePaste = (e) => {
    const pasted = e.clipboardData.getData("text").replace(/\D/g, "").slice(0, 6)
    if (!pasted) return
    e.preventDefault()
    const next = pasted.split("").concat(Array(6 - pasted.length).fill(""))
    setDigits(next)
    inputsRef.current[Math.min(pasted.length, 5)]?.focus()
  }

  // ---- Submit ----

  const handleSubmit = async (e) => {
    e.preventDefault()
    setError("")
    setInfo("")

    if (!email) {
      setError("Email is missing. Please go back and try again.")
      return
    }
    if (otp.length !== 6) {
      setError("Please enter the 6-digit code.")
      return
    }

    setLoading(true)

    if (mode === "reset") {
      // Password reset OTP — we don't complete the reset here; the
      // NewPasswordPage does that. We just carry the code forward.
      // For now, forward to /newpassword with email + otp in the URL.
      setLoading(false)
      navigate(`/newpassword?email=${encodeURIComponent(email)}&otp=${encodeURIComponent(otp)}`)
      return
    }

    // Registration OTP
    const result = await verifyOTP({ email, otp })
    setLoading(false)
    if (result.success) {
      navigate("/")
    } else {
      setError(result.error || "Invalid or expired code.")
    }
  }

  // ---- Resend ----

  const handleResend = async () => {
    setError("")
    setInfo("")
    if (!email) {
      setError("Email is missing.")
      return
    }

    setResending(true)
    try {
      if (mode === "reset") {
        const res = await api.post("/accounts/forgot-password/", { email })
        if (res.data?.success) setInfo("A new code has been sent to your email.")
        else setError(res.data?.error || "Could not resend code.")
      } else {
        const res = await api.post("/accounts/resend-otp/", { email })
        if (res.data?.success) setInfo("A new code has been sent to your email.")
        else setError(res.data?.error || "Could not resend code.")
      }
    } catch (err) {
      setError(err.response?.data?.error || "Could not resend code.")
    } finally {
      setResending(false)
    }
  }

  return (
    <main className="bg-white w-screen min-h-screen flex flex-col dark:bg-neutral-900 dark:text-white">
      <Scrolltotop />

      <div className="bg-white flex items-center sticky top-0 ps-4 py-4 shadow-md shadow-neutral-300 dark:bg-neutral-800">
        <button onClick={() => navigate(-1)} className="cursor-pointer">
          <FaArrowLeft />
        </button>
        <p className="text-lg font-bold absolute left-1/2 -translate-x-1/2">
          Verification
        </p>
      </div>

      <div className="flex flex-col justify-center gap-5 px-4 flex-1 py-8">
        <img src={verify} alt="" className="w-1/2 mx-auto" />

        <div className="text-center w-80 mx-auto mb-6 flex flex-col gap-3">
          <p className="text-orange-400 text-2xl font-bold">Verification Code</p>
          <p className="font-semibold leading-tight">
            Please enter the 6-digit code that was sent to
          </p>
          {email ? (
            <p className="text-sm text-neutral-600 dark:text-neutral-300 break-all">
              {email}
            </p>
          ) : (
            <p className="text-sm text-red-500">
              No email provided. Go back and try again.
            </p>
          )}
        </div>

        {error && (
          <p className="bg-red-100 text-red-600 border border-red-300 rounded-xl p-2 text-sm text-center">
            {error}
          </p>
        )}
        {info && (
          <p className="bg-green-100 text-green-700 border border-green-300 rounded-xl p-2 text-sm text-center">
            {info}
          </p>
        )}

        <form onSubmit={handleSubmit} className="flex flex-col gap-5">
          <div
            className="flex justify-between w-80 mx-auto gap-2"
            onPaste={handlePaste}
          >
            {digits.map((d, i) => (
              <input
                key={i}
                ref={(el) => (inputsRef.current[i] = el)}
                type="text"
                inputMode="numeric"
                maxLength={1}
                value={d}
                onChange={(e) => handleChange(i, e.target.value)}
                onKeyDown={(e) => handleKeyDown(i, e)}
                className="bg-orange-100 text-neutral-900 text-center text-xl font-bold outline-none w-12 h-12 rounded-xl focus:ring-2 focus:ring-orange-400"
              />
            ))}
          </div>

          <button
            type="submit"
            disabled={loading || otp.length !== 6}
            className="bg-gradient-to-b from-orange-600 to-orange-400 text-white rounded-xl py-2 cursor-pointer disabled:opacity-50"
          >
            {loading ? "Verifying..." : "Verify"}
          </button>

          <div className="flex flex-col items-center gap-1">
            <span className="text-sm text-neutral-500">Didn't receive the code?</span>
            <button
              type="button"
              onClick={handleResend}
              disabled={resending}
              className="text-orange-400 hover:text-orange-600 text-sm disabled:opacity-60"
            >
              {resending ? "Sending..." : "Resend code"}
            </button>
          </div>
        </form>
      </div>
    </main>
  )
}

export default VerificationPage
